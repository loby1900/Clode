require("dotenv").config(); const path = require("path"); const express = require("express"); const cors = require("cors"); const { randomUUID } = require("crypto"); const db = require("./db"); const { uploadsDir } = require("./middleware/upload"); const authRoutes = require("./routes/auth"); const postRoutes = require("./routes/posts"); const userRoutes = require("./routes/users"); const { encryptText, decryptText } = require("./crypto"); const { PROVIDER_CATALOG, normalizeProviderType, isAllowedProviderUrl, testProviderConnection, callProviderChat } = require("./providerService"); const { createProject, deleteProject, getProject, listAllFiles, listProjects, readProjectFile, writeProjectFile } = require("./projectService"); const app = express(); const PORT = Number(process.env.PORT || 3000); app.use(cors()); app.use(express.json({ limit: "5mb" })); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname, "..", "public"))); app.use("/uploads", express.static(uploadsDir)); app.use("/api/auth", authRoutes); app.use("/api/posts", postRoutes); app.use("/api/users", userRoutes); app.get("/api/health", (_req, res) => { res.json({ ok: true, app: "Clode AI Workbench", version: "1.0.0" }); }); app.get("/api/providers/catalog", (_req, res) => { res.json({ providers: PROVIDER_CATALOG }); }); app.get("/api/providers", (_req, res) => { const rows = db.prepare("SELECT * FROM ai_provider_connections ORDER BY updated_at DESC").all(); const sanitizedRows = rows.map((row) => ({ ...row, api_key: undefined, api_key_cipher: undefined, apiKey: null })); res.json({ providers: sanitizedRows }); }); app.post("/api/providers", async (req, res) => { try { const payload = req.body || {}; const providerType = normalizeProviderType(payload.providerType || payload.provider_type || "openai-compatible"); const name = String(payload.name || payload.connectionLabel || "Custom provider").trim(); const baseUrl = String(payload.baseUrl || payload.base_url || "").trim(); const modelId = payload.modelId ? String(payload.modelId).trim() : payload.model_id ? String(payload.model_id).trim() : ""; const apiKey = payload.apiKey ? String(payload.apiKey).trim() : payload.api_key ? String(payload.api_key).trim() : ""; const allowLocal = Boolean(payload.allowLocal || payload.allow_local); if (!name || !baseUrl) { return res.status(400).json({ error: "Provider name and base URL are required." }); } if (!isAllowedProviderUrl(baseUrl, allowLocal)) { return res.status(400).json({ error: "Base URL is invalid or blocked. Use a trusted URL; allow local endpoints only when needed." }); } const id = String(payload.id || randomUUID()); const safeApiKey = apiKey ? encryptText(apiKey) : null; db.prepare(` INSERT INTO ai_provider_connections (id, name, provider_type, base_url, api_key_cipher, model_id, connection_label, allow_local, updated_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP) ON CONFLICT(id) DO UPDATE SET name = excluded.name, provider_type = excluded.provider_type, base_url = excluded.base_url, api_key_cipher = excluded.api_key_cipher, model_id = excluded.model_id, connection_label = excluded.connection_label, allow_local = excluded.allow_local, updated_at = CURRENT_TIMESTAMP `).run(id, name, providerType, baseUrl, safeApiKey, modelId || null, name, allowLocal ? 1 : 0); res.status(201).json({ ok: true, provider: { id, name, providerType, baseUrl, modelId, allowLocal } }); } catch (error) { res.status(500).json({ error: error.message || "Unable to save provider settings." }); } }); app.post("/api/providers/test", async (req, res) => { try { const config = req.body || {}; const result = await testProviderConnection(config); res.status(result.status === "error" ? 400 : 200).json(result); } catch (error) { res.status(500).json({ status: "error", code: "TEST_FAILED", message: error.message || "Connection test failed." }); } }); app.delete("/api/providers/:id", (req, res) => { const { id } = req.params; const row = db.prepare("SELECT id FROM ai_provider_connections WHERE id = ?").get(id); if (!row) return res.status(404).json({ error: "Provider not found." }); db.prepare("DELETE FROM ai_provider_connections WHERE id = ?").run(id); res.json({ ok: true, deleted: id }); }); app.get("/api/projects", (_req, res) => { res.json({ projects: listProjects() }); }); app.post("/api/projects", (req, res) => { try { const { name, description } = req.body || {}; const project = createProject({ name, description }); res.status(201).json({ project }); } catch (error) { res.status(400).json({ error: error.message || "Unable to create project." }); } }); app.delete("/api/projects/:id", (req, res) => { try { const result = deleteProject(req.params.id); res.json(result); } catch (error) { res.status(400).json({ error: error.message || "Unable to delete project." }); } }); app.get("/api/projects/:id/files", (req, res) => { try { const project = getProject(req.params.id); if (!project) return res.status(404).json({ error: "Project not found." }); res.json({ project, files: listAllFiles(project.root_path) }); } catch (error) { res.status(400).json({ error: error.message || "Unable to list project files." }); } }); app.get("/api/projects/:id/file", (req, res) => { try { const filePath = req.query.path || ""; const content = readProjectFile(req.params.id, filePath); res.json({ path: filePath, content }); } catch (error) { res.status(400).json({ error: error.message || "Unable to read project file." }); } }); app.post("/api/projects/:id/file", (req, res) => { try { const filePath = req.body && req.body.path ? req.body.path : ""; const content = req.body && Object.prototype.hasOwnProperty.call(req.body, "content") ? req.body.content : ""; const result = writeProjectFile(req.params.id, filePath, content); res.json({ ok: true, file: result }); } catch (error) { res.status(400).json({ error: error.message || "Unable to save project file." }); } }); app.post("/api/chat", async (req, res) => { try { const { message, projectId, providerId, providerType, modelId } = req.body || {}; if (!message || !String(message).trim()) { return res.status(400).json({ error: "Message cannot be empty." }); } const rows = db.prepare("SELECT * FROM ai_provider_connections ORDER BY updated_at DESC").all(); const provider = providerId ? rows.find((row) => row.id === providerId) : rows[0]; if (!provider) { return res.status(400).json({ error: "No provider is configured. Add a provider under Integrations before sending a chat request.", requiresConfig: true }); } const apiKey = provider.api_key_cipher ? decryptText(provider.api_key_cipher) : ""; const config = { id: provider.id, name: provider.name, providerType: normalizeProviderType(provider.provider_type || providerType || "openai-compatible"), baseUrl: provider.base_url, apiKey, modelId: provider.model_id || modelId || "gpt-4o-mini", allowLocal: Boolean(provider.allow_local) }; const project = projectId ? getProject(projectId) : null; let promptContext = ""; if (project) { const fileList = listAllFiles(project.root_path).slice(0, 15).map((item) => item.path).join("\n"); promptContext = `Project: ${project.name}\nFiles:\n${fileList || "No files yet."}`; } const response = await callProviderChat(config, [ { role: "system", content: "You are Clode, a secure project-aware AI engineering assistant. Be clear, honest, and explicit about required configuration or missing capabilities. Never pretend a feature exists without real provider support." }, { role: "user", content: `${promptContext}\n\nUser request:\n${message}` } ]); res.json({ ok: true, answer: response.content, provider: response.provider, model: response.model, usage: response.usage || null }); } catch (error) { res.status(500).json({ error: error.message || "Chat request failed." }); } }); app.get("/api/health/provider", async (_req, res) => { const rows = db.prepare("SELECT * FROM ai_provider_connections ORDER BY updated_at DESC LIMIT 1").all(); res.json({ configured: rows.length > 0, provider: rows[0] || null }); }); app.get("*", (req, res, next) => { if (req.path.startsWith("/api") || req.path.startsWith("/uploads")) return next(); res.sendFile(path.join(__dirname, "..", "public", "index.html")); }); app.use((err, _req, res, _next) => { if (err) { const status = err.status || 500; res.status(status).json({ error: err.message || "Unexpected server error." }); } }); app.listen(PORT, () => { console.log(`Clode AI Workbench running at http://localhost:${PORT}`); });
